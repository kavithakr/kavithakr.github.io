def compute_loss(y_true, y_pred):
    # mean squared error
    n = len(y_true)
    return (1 / n) * np.sum((y_pred - y_true) ** 2)

def train_linear_regression(X, y, learning_rate, num_iterations):
    n_samples = X.shape[0]
    weight = 0
    bias = 0
    weight_hist = []
    bias_hist = []
    plt.figure(figsize=(10, 6))
    x_plot = np.linspace(X.min() - 1, X.max() + 1, 100)

    plt.scatter(X, y, color='red', marker='o', label='Data Points')
    plt.xlabel('X')
    plt.ylabel('Y')
    plt.title('Regression Line during Training')
    plt.legend()
    plt.grid(True)

    for iter in range(num_iterations):
        y_predicted = weight * X + bias

        dw = (2 / n_samples) * np.sum(X * (y_predicted - y))
        db = (2 / n_samples) * np.sum(y_predicted - y)

        weight -= learning_rate * dw
        bias -= learning_rate * db
        weight_hist.append(weight)
        bias_hist.append(bias)

        plot_iterations = [0, 1, 10000, 25000, 40000, 50000, 60000, 80000, 99999]

        if iter in plot_iterations:
            current_weight = weight
            current_bias = bias
            current_line = current_weight * x_plot + current_bias
            plt.plot(x_plot, current_line, label=f'Iteration {iter}')

    plt.legend()
    plt.grid(True)
    plt.show()

    return weight_hist, bias_hist
